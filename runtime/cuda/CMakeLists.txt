cmake_minimum_required(VERSION 3.18)
project(nanhu_cuda LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NANHU_ENABLE_CUDA "Build the actual GPU backend" ON)
add_executable(nanhu_gpu src/main.cpp src/engine_cpu.cpp src/memory_plan.cpp)
target_include_directories(nanhu_gpu PRIVATE src vendor)
add_executable(op_test tests/op_test.cpp src/engine_cpu.cpp src/memory_plan.cpp)
target_include_directories(op_test PRIVATE src tests)
add_executable(memory_direct_test tests/memory_direct_test.cpp src/engine_cpu.cpp)
target_include_directories(memory_direct_test PRIVATE src)
if(NANHU_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "RTX2060 compute capability")
  endif()
  enable_language(CUDA)
  foreach(t nanhu_gpu op_test memory_direct_test)
    target_sources(${t} PRIVATE src/engine_cuda.cu)
    target_compile_definitions(${t} PRIVATE NANHU_HAS_CUDA=1)
    set_target_properties(${t} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(${t} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
  endforeach()
  target_sources(memory_direct_test PRIVATE src/memory_plan.cpp)
endif()
enable_testing()
add_test(NAME cpu_schedule COMMAND nanhu_gpu --cpu --self-test)
add_test(NAME cpu_opcodes COMMAND op_test --cpu)
add_executable(memory_plan_test tests/memory_plan_test.cpp src/memory_plan.cpp src/engine_cpu.cpp)
target_include_directories(memory_plan_test PRIVATE src)
add_test(NAME cpu_memory_plan COMMAND memory_plan_test)
add_test(NAME cpu_direct_memory COMMAND memory_direct_test --cpu)
add_executable(flat_plan_test tests/flat_plan_test.cpp src/flat_ops.cpp src/engine_cpu.cpp)
target_include_directories(flat_plan_test PRIVATE src)
add_test(NAME cpu_flat_plan COMMAND flat_plan_test)
foreach(t nanhu_gpu op_test memory_direct_test memory_plan_test)
  target_sources(${t} PRIVATE src/flat_ops.cpp)
endforeach()
add_test(NAME cpu_flat_schedule COMMAND nanhu_gpu --cpu --self-test --flat-descriptors)
add_test(NAME cpu_flat_opcodes COMMAND op_test --cpu --flat-descriptors)
add_test(NAME cpu_flat_memory COMMAND memory_direct_test --cpu --flat-descriptors)
if(NANHU_ENABLE_CUDA)
  add_test(NAME gpu_opcodes COMMAND op_test)
  add_test(NAME gpu_graph_tail COMMAND nanhu_gpu --self-test)
  add_test(NAME gpu_graph_layers COMMAND nanhu_gpu --self-test --tail-threshold 0)
  add_test(NAME gpu_stream_tail COMMAND nanhu_gpu --self-test --no-graph)
  add_test(NAME gpu_candidates_opcodes COMMAND op_test --memory-candidates)
  add_test(NAME gpu_candidates_graph_tail COMMAND nanhu_gpu --self-test --memory-candidates)
  add_test(NAME gpu_candidates_graph_layers COMMAND nanhu_gpu --self-test --memory-candidates --tail-threshold 0)
  add_test(NAME gpu_candidates_stream_tail COMMAND nanhu_gpu --self-test --memory-candidates --no-graph)
  add_test(NAME gpu_direct_memory_graph_tail COMMAND memory_direct_test)
  add_test(NAME gpu_direct_memory_graph_layers COMMAND memory_direct_test --tail-threshold 0 --threads 128)
  add_test(NAME gpu_direct_memory_stream_tail COMMAND memory_direct_test --no-graph --threads 128)
  add_test(NAME gpu_flat_opcodes COMMAND op_test --flat-descriptors)
  add_test(NAME gpu_flat_graph_tail COMMAND nanhu_gpu --self-test --flat-descriptors)
  add_test(NAME gpu_flat_graph_layers COMMAND nanhu_gpu --self-test --flat-descriptors --tail-threshold 0)
  add_test(NAME gpu_flat_stream_tail COMMAND nanhu_gpu --self-test --flat-descriptors --no-graph)
  add_test(NAME gpu_flat_stream_layers COMMAND nanhu_gpu --self-test --flat-descriptors --no-graph --tail-threshold 0)
  add_test(NAME gpu_flat_candidates_opcodes COMMAND op_test --flat-descriptors --memory-candidates)
  add_test(NAME gpu_flat_candidates_graph_tail COMMAND nanhu_gpu --self-test --flat-descriptors --memory-candidates)
  add_test(NAME gpu_flat_candidates_graph_layers COMMAND nanhu_gpu --self-test --flat-descriptors --memory-candidates --tail-threshold 0)
  add_test(NAME gpu_flat_direct_memory_graph_tail COMMAND memory_direct_test --flat-descriptors)
  add_test(NAME gpu_flat_direct_memory_graph_layers COMMAND memory_direct_test --flat-descriptors --tail-threshold 0 --threads 128)
  add_test(NAME gpu_flat_direct_memory_stream_tail COMMAND memory_direct_test --flat-descriptors --no-graph --threads 128)
endif()

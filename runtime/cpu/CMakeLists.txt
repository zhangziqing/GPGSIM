cmake_minimum_required(VERSION 3.18)
project(nanhu_cpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(yaml-cpp CONFIG REQUIRED)
if(TARGET yaml-cpp::yaml-cpp)
  set(NANHU_YAML_TARGET yaml-cpp::yaml-cpp)
else()
  set(NANHU_YAML_TARGET yaml-cpp)
endif()
get_filename_component(ARTIFACT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(RTEAAL_SIM_ROOT "${ARTIFACT_ROOT}/RTeAAL-Sim" CACHE PATH "Pinned simulator source")
set(NANHU_MODEL_ROOT "${ARTIFACT_ROOT}/build/model" CACHE PATH "Generated tensors and metadata")
option(NANHU_NATIVE "Use the measured -march=native host setting" ON)
set(KERNEL_ROOT "${RTEAAL_SIM_ROOT}/kernel")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${ARTIFACT_ROOT}/tools/generate_nu_header.py"
  "${CMAKE_CURRENT_BINARY_DIR}/kernel.h" RESULT_VARIABLE header_status)
if(NOT header_status EQUAL 0)
  message(FATAL_ERROR "Cannot generate the pinned NU evaluator")
endif()
set(FESVR_SOURCES elfloader.cc htif.cc memif.cc dtm.cc syscall.cc device.cc rfb.cc
  context.cc htif_pthread.cc htif_hexwriter.cc dummy.cc option_parser.cc term.cc tsi.cc)
list(TRANSFORM FESVR_SOURCES PREPEND "${KERNEL_ROOT}/")
add_library(nanhu_fesvr STATIC ${FESVR_SOURCES})
target_include_directories(nanhu_fesvr PUBLIC "${KERNEL_ROOT}")
target_link_libraries(nanhu_fesvr PUBLIC Threads::Threads)
target_compile_definitions(nanhu_fesvr PRIVATE PREFIX="${CMAKE_CURRENT_BINARY_DIR}")
function(nanhu_host target source)
  add_executable(${target} "${source}")
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}"
    "${KERNEL_ROOT}" "${YAML_CPP_INCLUDE_DIR}" "${ARTIFACT_ROOT}/runtime/common" "${ARTIFACT_ROOT}/runtime/cuda/vendor")
  target_compile_definitions(${target} PRIVATE NANHU_LARGE_LAYERS
    NANHU_TENSOR_ROOT="${NANHU_MODEL_ROOT}/kernel" NANHU_METADATA_ROOT="${NANHU_MODEL_ROOT}/utils/bin"
    NANHU_LAYOUT_FILE="${NANHU_MODEL_ROOT}/layout.json" COREMARK_DIR="${ARTIFACT_ROOT}/configs"
    NANHU_TUNED_HEADER="kernel.h")
  target_compile_options(${target} PRIVATE -O3 -g)
  if(NANHU_NATIVE)
    target_compile_options(${target} PRIVATE -march=native)
  endif()
  target_link_libraries(${target} PRIVATE nanhu_fesvr ${NANHU_YAML_TARGET})
endfunction()
nanhu_host(checked_loop checked_loop.cc)
nanhu_host(coremark rteaal_coremark.cc)
nanhu_host(coremark_thp rteaal_coremark_thp.cc)
nanhu_host(coremark_thp_prefault rteaal_coremark_thp_prefault.cc)
nanhu_host(large_layers_test tests/large_layers.cc)
nanhu_host(grouped_test tests/grouped.cc)
add_executable(preprocess_tensor "${KERNEL_ROOT}/preprocess_tensor.cc")
target_include_directories(preprocess_tensor PRIVATE "${ARTIFACT_ROOT}/runtime/cuda/vendor" "${YAML_CPP_INCLUDE_DIR}")
target_compile_definitions(preprocess_tensor PRIVATE JSONPATH="${NANHU_MODEL_ROOT}/kernel/json/"
  TXTPATH="${NANHU_MODEL_ROOT}/kernel/txt/" YAMLPATH="${NANHU_MODEL_ROOT}/kernel/yaml/")
target_link_libraries(preprocess_tensor PRIVATE ${NANHU_YAML_TARGET} Threads::Threads)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/model-root.txt" "${NANHU_MODEL_ROOT}\n")
enable_testing()
add_test(NAME ru_wide_layer_counter COMMAND large_layers_test)
add_test(NAME nu_grouped_against_ou COMMAND grouped_test)
